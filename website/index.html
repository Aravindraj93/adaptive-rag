from adaptive_rag import BM25Retriever, Chunk, SemanticCachedRetriever

# Zero dependencies — runs anywhere
backend = BM25Retriever()
backend.add([
    Chunk("policy-1", "docs", "Refunds available within 30 days."),
    Chunk("policy-2", "docs", "Shipping takes 3-5 business days."),
])

# Semantic caching: repeated queries return instantly
retriever = SemanticCachedRetriever(
    backend, scope="my-app", revision=lambda: 1
)

results = retriever.search("refund period", top_k=3)
for r in results:
    print(r.chunk.text)
from adaptive_rag.integrations.langchain import AdaptiveRetriever
from langchain.chains import RetrievalQA

# From LangChain documents — 1 line change
retriever = AdaptiveRetriever.from_langchain_docs(
    documents,
    scope="my-app",
    use_cache=True,
)

# Drop into any LangChain chain
chain = RetrievalQA.from_chain_type(
    llm=llm,
    retriever=retriever.as_langchain_retriever(),
)
response = chain.invoke({"query": "What is the return policy?"})
from adaptive_rag.integrations.llamaindex import AdaptiveQueryEngine

# From LlamaIndex documents — replaces VectorStoreIndex
engine = AdaptiveQueryEngine.from_documents(
    documents,
    scope="my-app",
    llm=llm,  # optional — for answer synthesis
)

# Same API as LlamaIndex query engines
response = engine.query("What are the pricing tiers?")
print(response.response)
for node in response.source_nodes:
    print(node.node.text[:100])
from adaptive_rag.packs import HealthcareRAG

# HIPAA-safe — zero network calls, all on CPU
rag = HealthcareRAG()
rag.add_documents("./clinical_guidelines/")

# Medical abbreviation expansion built-in
# "htn" → "hypertension", "bp" → "blood pressure"
results = rag.search("first-line treatment htn", top_k=5)
for r in results:
    print(r.chunk.text)

print(rag.info())
